How the code works

// 1. Call Grafana's query API (proxied via nginx to avoid CORS) const res = await fetch('/grafana-api/api/ds/query', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ queries: [{ refId: 'A', datasource: { uid: 'mongodb-api-ds' }, type: 'json', source: 'url', url: 'http://mongo-api:4000/aggregate/orders', url_options: { method: 'POST', data: '{"pipeline":[{"$group":{"_id":"$status","count":{"$sum":1}}}]}' }, columns: [ { selector: '_id', text: 'Status', type: 'string' }, { selector: 'count', text: 'Count', type: 'number' } ] }], from: 'now-1h', to: 'now' }) }); // 2. Parse Grafana data frames format const json = await res.json(); const frame = json.results.A.frames[0]; const labels = frame.data.values[0]; // string column const counts = frame.data.values[1]; // number column // 3. Hand off to Chart.js (or Highcharts) new Chart(canvas, { type: 'bar', data: { labels, datasets: [{ data: counts }] } });

Pros

  • No Grafana alerting on these specific panels

Stock chart code

// 1. Call mongo-api directly — Grafana not involved const res = await fetch('http://localhost:4000/aggregate/products', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ pipeline: [{ $project: { _id: 0, name: 1, stock: 1 } }] }) }); // 2. Plain JSON array — no special parsing needed const data = await res.json(); const labels = data.map(d => d.name); const values = data.map(d => d.stock); // 3. Rounded bars with borderRadius: 8 new Chart(canvas, { type: 'bar', data: { labels, datasets: [{ data: values, borderRadius: 8, borderSkipped: false }] } });

Revenue trend code

// Revenue line chart with gradient fill const res = await fetch('http://localhost:4000/aggregate/daily_revenue', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ pipeline: [ { $sort: { date: 1 } }, { $project: { _id: 0, day: 1, revenue: 1 } } ] }) }); const data = await res.json(); // Create canvas gradient fill const grad = ctx.createLinearGradient(0,0,0,300); grad.addColorStop(0, 'rgba(16,185,129,.35)'); grad.addColorStop(1, 'rgba(16,185,129,.02)'); new Chart(canvas, { type: 'line', data: { labels: data.map(d=>d.day), datasets: [{ data: data.map(d=>d.revenue), fill: true, backgroundColor: grad, tension: 0.4 }] } });
